import assert from 'node:assert/strict';
import test from 'node:test';
import { createClient, initializeEdge, response, uuid } from './helpers.mjs';

const ATTEMPT = uuid(7000), USER = uuid(101), GENERATION = uuid(9000), LEASE = uuid(8000);
const MARKER = 'SYNTHETIC_PRIVATE_PROVIDER_MARKER';
const serviceKey = 'synthetic-service-key';
function rows(length = 1) {
  return Array.from({ length }, (_, index) => ({
    id: uuid(1000 + index), user_id: USER, token: `synthetic-private-push-token-${index}`,
    registration_generation: GENERATION,
  }));
}
function payload(changes = {}) {
  return { request_id: ATTEMPT, type: 'assignment', record: {
    assignee_id: USER, label: 'Synthetic draft', date: '2099-01-01', time: '07:00',
  }, old_record: { assignee_id: null }, ...changes };
}
function request(name, body, authorization = `Bearer ${serviceKey}`, method = 'POST') {
  return new Request(`https://synthetic-push-receipts.invalid/functions/v1/${name}`, {
    method, headers: { authorization, 'Content-Type': 'application/json' },
    ...(method === 'POST' ? { body: JSON.stringify(body ?? {}) } : {}),
  });
}
function deferred() {
  let resolve;
  const promise = new Promise(done => { resolve = done; });
  return { promise, resolve };
}
async function fixture(options = {}) {
  const databaseCalls = [], providerCalls = [], logs = [], trace = [], handlers = {}, attempts = new Map();
  const state = { tokens: options.tokens ?? rows(), claimItems: options.claimItems ?? [], ticketSequence: 20000 };
  const database = async (input, init) => {
    const url = new URL(String(input));
    assert.equal(url.host, 'synthetic-push-receipts.invalid');
    assert.equal(new Headers(init.headers).get('apikey'), serviceKey);
    const name = url.pathname.replace('/rest/v1/', '');
    const params = init.body ? JSON.parse(init.body) : null;
    databaseCalls.push({ name, params, method: init.method, url: url.toString(), signal: init.signal });
    trace.push(`db:${name}`);
    if (name.startsWith('rpc/')) {
      const rpc = name.slice(4);
      const intercepted = await options.rpc?.(rpc, params, { attempts, state, trace, signal: init.signal });
      if (intercepted !== undefined) return intercepted;
      if (rpc === 'push_receipts_prepare') {
        assert.equal(init.method, 'POST');
        if (attempts.has(params.p_attempt_id)) return response({ attempt_id: params.p_attempt_id, started: false });
        attempts.set(params.p_attempt_id, params.p_registrations.map(entry => ({ ...entry, status: 'unknown', ticket_id: null, error_code: null })));
        return response({ attempt_id: params.p_attempt_id, started: true });
      }
      if (rpc === 'push_receipts_record_tickets') {
        const slots = attempts.get(params.p_attempt_id);
        assert.ok(slots, 'Durable prepare must precede ticket recording');
        for (const result of params.p_results) Object.assign(slots[result.position], result);
        return response({ attempt_id: params.p_attempt_id, recorded_count: params.p_results.length });
      }
      if (rpc === 'push_receipts_claim_due') return response(options.claimResult ?? { lease_id: LEASE, items: state.claimItems });
      if (rpc === 'push_receipts_apply') {
        const results = params.p_results;
        return response(options.applyResult ?? {
          lease_id: params.p_lease_id, applied_count: results.length,
          receipt_ok_count: results.filter(entry => entry.status === 'receipt_ok').length,
          receipt_error_count: results.filter(entry => entry.status === 'receipt_error').length,
          missing_count: results.filter(entry => entry.status === 'missing').length,
          deleted_token_count: options.deletedTokenCount ?? 0,
        });
      }
      throw new Error('Unexpected RPC in synthetic fixture');
    }
    assert.equal(init.method, 'GET', 'Service must never directly mutate token/profile rows');
    if (options.lookupError === name) return response({ code: MARKER, message: MARKER, details: MARKER }, 503);
    if (name === 'push_tokens') {
      assert.equal(url.searchParams.get('select'), 'id,user_id,token,registration_generation');
      return response(state.tokens);
    }
    if (name === 'notification_preferences') return response([]);
    if (name === 'profiles') return response({ full_name: 'Synthetic author' });
    if (name === 'conversation_members') return response([{ user_id: USER }]);
    if (name === 'conversations') return response(options.conversation ?? { is_group: false, stable_id: null });
    if (name === 'stable_members') return response([{ user_id: USER }]);
    if (name === 'blocked_users') return response([]);
    throw new Error('Unexpected table in synthetic fixture');
  };
  const client = createClient('https://synthetic-push-receipts.invalid', serviceKey, {
    auth: { persistSession: false, autoRefreshToken: false, detectSessionInUrl: false },
    global: { fetch: database },
  });
  const provider = async (input, init) => {
    const url = new URL(String(input));
    assert.equal(url.host, 'exp.host');
    assert.ok(['/--/api/v2/push/send', '/--/api/v2/push/getReceipts'].includes(url.pathname));
    const body = JSON.parse(init.body);
    providerCalls.push({ url: url.toString(), body, signal: init.signal });
    trace.push(`provider:${url.pathname}`);
    if (options.provider) return options.provider(url, body, { attempts, state, trace, providerCalls });
    return url.pathname.endsWith('/send')
      ? response({ data: body.map(() => ({ status: 'ok', id: uuid(state.ticketSequence++) })) })
      : response({ data: Object.fromEntries(body.ids.map(id => [id, { status: 'ok' }])) });
  };
  for (const name of ['send-push-notification', 'collect-push-receipts']) {
    await initializeEdge(name, {
      createClient: () => client,
      Deno: { env: { get: key => ({ SUPABASE_URL: 'https://synthetic-push-receipts.invalid', SUPABASE_SERVICE_ROLE_KEY: options.envKey ?? serviceKey })[key] }, serve: handler => { handlers[name] = handler; } },
      fetch: provider, crypto: { randomUUID: () => uuid(6000) },
      console: { error: (...args) => logs.push(args) },
    });
  }
  return {
    state, attempts, databaseCalls, providerCalls, logs, trace,
    send: (body = payload(), auth, method) => handlers['send-push-notification'](request('send-push-notification', body, auth, method)),
    collect: (body = {}, auth, method) => handlers['collect-push-receipts'](request('collect-push-receipts', body, auth, method)),
  };
}
function noLeaks(fixture) {
  assert.doesNotMatch(JSON.stringify(fixture.logs), /SYNTHETIC_PRIVATE_PROVIDER_MARKER|synthetic-private-push-token|synthetic-service-key/);
  assert.ok(fixture.logs.every(([prefix]) => /^\[push (notification|receipts)\]/.test(prefix)));
}

test('actual SDK reserves all 101 generation snapshots before first HTTP and sends only original message fields', async () => {
  const f = await fixture({ tokens: rows(101), provider: async (_url, body, context) => {
    assert.equal(context.attempts.get(ATTEMPT).length, 101);
    assert.deepEqual(Object.keys(body[0]).sort(), ['body', 'channelId', 'data', 'sound', 'title', 'to']);
    return response({ data: body.map(() => ({ status: 'ok', id: uuid(context.state.ticketSequence++) })) });
  } });
  const result = await f.send();
  assert.deepEqual(await result.json(), { sent: 101 });
  assert.deepEqual(f.providerCalls.map(call => call.body.length), [100, 1]);
  const prepare = f.databaseCalls.find(call => call.name === 'rpc/push_receipts_prepare').params;
  assert.equal(prepare.p_attempt_id, ATTEMPT);
  assert.equal(prepare.p_registrations.length, 101);
  assert.deepEqual(prepare.p_registrations[0], { position: 0, token_id: uuid(1000), user_id: USER, registration_generation: GENERATION });
  assert.doesNotMatch(JSON.stringify(prepare), /push-token|Synthetic draft|2099-01-01/);
  assert.deepEqual(f.trace.slice(2), ['db:rpc/push_receipts_prepare', 'provider:/--/api/v2/push/send', 'db:rpc/push_receipts_record_tickets', 'provider:/--/api/v2/push/send', 'db:rpc/push_receipts_record_tickets']);
});

test('mixed first chunk preserves every accepted ticket before failure and leaves unsent second chunk unknown', async () => {

    : index === 2 ? null : { status: 'ok', id: uuid(30000 + index) }) }) });
  assert.equal((await f.send()).status, 500);
  const slots = f.attempts.get(ATTEMPT);
  assert.equal(slots.filter(slot => slot.status === 'accepted').length, 98);

  assert.equal(slots[100].status, 'unknown');
  assert.equal(f.providerCalls.length, 1);
  assert.deepEqual(await (await f.send()).json(), { sent: 0 });
  assert.equal(f.providerCalls.length, 1);
  noLeaks(f);
});

test('concurrent retry while first provider result is pending has one HTTP and immutable registration snapshot', async () => {
  const gate = deferred(), reached = deferred();
  const f = await fixture({ provider: async (_url, _body, context) => {
    reached.resolve();
    await gate.promise;
    assert.equal(context.attempts.get(ATTEMPT)[0].registration_generation, GENERATION);
    return response({ data: [{ status: 'ok', id: 'synthetic-pending-ticket' }] });

test('actual SDK and provider waits end after 15s; late responses cannot send, acknowledge or delete', { timeout: 20000 }, async () => {
  const prepareGate = deferred(), sendGate = deferred(), receiptGate = deferred();
  const gates = [prepareGate, sendGate, receiptGate];
  let firstPrepare = true;
  const preparing = await fixture({ rpc: async (name, params, context) => {
    if (name !== 'push_receipts_prepare' || !firstPrepare) return undefined;
    firstPrepare = false;
    context.attempts.set(ATTEMPT, params.p_registrations.map(entry => ({ ...entry, status: 'unknown' })));
    await prepareGate.promise; // Synthetic transport deliberately ignores abort.
    return response({ attempt_id: ATTEMPT, started: true });
  } });
  const sending = await fixture({ provider: async () => {
    await sendGate.promise;
    return response({ data: [{ status: 'ok', id: 'synthetic-late-send-ticket' }] });
  } });
  const collecting = await fixture({ claimItems: [claimItems[1]], provider: async () => {
    const reply = response({});
    reply.json = async () => {
      await receiptGate.promise;
      return { data: { 'synthetic-ticket-dnr': { status: 'error', details: { error: 'DeviceNotRegistered' } } } };
    };
    return reply;
  } });
  const startedAt = performance.now();
  const pending = [preparing.send(), sending.send(), collecting.collect()];
  try {
    const results = await Promise.all(pending);
    assert.ok(performance.now() - startedAt >= 14500);
    assert.ok(performance.now() - startedAt < 19000);
    assert.deepEqual(results.map(result => result.status), [500, 500, 500]);
    assert.equal(preparing.providerCalls.length, 0);
    assert.equal(preparing.databaseCalls.find(call => call.name === 'rpc/push_receipts_prepare').signal.aborted, true);
    assert.equal(sending.providerCalls[0].signal.aborted, true);
    assert.equal(collecting.providerCalls[0].signal.aborted, true);
    assert.ok(!sending.databaseCalls.some(call => call.name === 'rpc/push_receipts_record_tickets'));
    assert.deepEqual(collecting.databaseCalls.find(call => call.name === 'rpc/push_receipts_apply').params.p_results,
      [{ ...claimItems[1], status: 'missing', error_code: null }]);
    assert.deepEqual(await (await preparing.send()).json(), { sent: 0 });
    assert.deepEqual(await (await sending.send()).json(), { sent: 0 });
    for (const gate of gates) gate.resolve();
    await new Promise(resolve => setTimeout(resolve, 20));
    assert.equal(preparing.providerCalls.length, 0);
    assert.equal(sending.providerCalls.length, 1);
    assert.equal(collecting.providerCalls.length, 1);
    assert.ok(!sending.databaseCalls.some(call => call.name === 'rpc/push_receipts_record_tickets'));
    assert.equal(collecting.databaseCalls.filter(call => call.name === 'rpc/push_receipts_apply').length, 1);
    for (const f of [preparing, sending, collecting]) noLeaks(f);
  } finally {
    for (const gate of gates) gate.resolve();
    await Promise.allSettled(pending);
  }
});

    ['push_tokens', payload()], ['notification_preferences', payload()], ['conversation_members', message],
    ['conversations', message], ['stable_members', message], ['blocked_users', message], ['stable_members', alert],
  ]) {
    const f = await fixture({ lookupError, conversation: { is_group: true, stable_id: uuid(202) } });

  }
});
